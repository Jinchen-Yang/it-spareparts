## 0. 启动同一个受控发布 shell

先由发布负责人把完整的、已批准的 40 位 SHA 放入环境变量，再进入同一个维护 shell。不要把
token、密码或 `.env` 内容写进状态文件。

```bash

  printf 'FATAL: %s\n' "$*" >&2
  exit 1

  sudo docker compose exec -T db \
    psql -U spareparts -d spareparts -v ON_ERROR_STOP=1 "$@"

    psql -U spareparts -d postgres -v ON_ERROR_STOP=1 "$@"

[[ "${TARGET_COMMIT:-}" =~ ^[0-9a-f]{40}$ ]] \
```

## 1. 固定旧基线，并从精确 SHA 的干净 archive 构建

### 1.1 记录正在运行的精确旧镜像与数据库 head

镜像 ID 必须来自当前运行容器的 `.Image`，镜像引用必须来自当前容器的 `.Config.Image`。
不要用 `docker compose config --images | awk '/app/'`：该输出没有稳定的服务名映射。

```bash
git fetch origin main
test -z "$(git status --porcelain --untracked-files=no)" \
  || fatal "生产仓库存在 tracked 改动"
test "$(git rev-parse "${TARGET_COMMIT}^{commit}")" = "$TARGET_COMMIT"
git merge-base --is-ancestor "$TARGET_COMMIT" origin/main \
  || fatal "TARGET_COMMIT 不属于 origin/main"

OLD_COMMIT=$(git rev-parse HEAD)
OLD_APP_CID=$(sudo docker compose ps -q app)
OLD_FRONTEND_CID=$(sudo docker compose ps -q frontend)
test -n "$OLD_APP_CID"
test -n "$OLD_FRONTEND_CID"
test "$(sudo docker inspect -f '{{.State.Running}}' "$OLD_APP_CID")" = "true"
test "$(sudo docker inspect -f '{{.State.Running}}' "$OLD_FRONTEND_CID")" = "true"

OLD_APP_IMAGE_ID=$(sudo docker inspect -f '{{.Image}}' "$OLD_APP_CID")
OLD_FRONTEND_IMAGE_ID=$(sudo docker inspect -f '{{.Image}}' "$OLD_FRONTEND_CID")
APP_IMAGE_REF=$(sudo docker inspect -f '{{.Config.Image}}' "$OLD_APP_CID")
FRONTEND_IMAGE_REF=$(sudo docker inspect -f '{{.Config.Image}}' "$OLD_FRONTEND_CID")
[[ "$OLD_APP_IMAGE_ID" == sha256:* ]]
[[ "$OLD_FRONTEND_IMAGE_ID" == sha256:* ]]
[[ -n "$APP_IMAGE_REF" && "$APP_IMAGE_REF" != sha256:* && "$APP_IMAGE_REF" != *@* ]]
[[ -n "$FRONTEND_IMAGE_REF" && "$FRONTEND_IMAGE_REF" != sha256:* && "$FRONTEND_IMAGE_REF" != *@* ]]

test "$(sudo docker image inspect -f '{{.Id}}' "$FRONTEND_IMAGE_REF")" = "$OLD_FRONTEND_IMAGE_ID"

save_state OLD_COMMIT "$OLD_COMMIT"
```

### 1.2 生成可验明提交来源的 archive，并只从该目录构建

构建上下文不得使用生产 checkout。`git archive` 不包含工作区脏文件、未跟踪文件或另一个
提交的内容；`git get-tar-commit-id` 必须反解回目标 SHA。

```bash
SOURCE_TAR="$PWD/backups/v1.19-source-${TARGET_COMMIT}.tar"
BUILD_PROJECT="v119build${TARGET_COMMIT:0:12}"
RELEASE_SRC=$(mktemp -d "$PWD/backups/.v1.19-src.XXXXXX")
chmod 700 "$RELEASE_SRC"

cleanup_build_artifacts() {
  sudo docker compose \
    --project-name "$BUILD_PROJECT" \
    --env-file "$PROD_ENV" \
    --project-directory "$RELEASE_SRC" \
    -f "$RELEASE_SRC/docker-compose.yml" \
    down --volumes --remove-orphans >/dev/null 2>&1 || true
  if [[ -d "$RELEASE_SRC" ]]; then
    case "$RELEASE_SRC" in
      "$PWD"/backups/.v1.19-src.*)
        find "$RELEASE_SRC" -depth -mindepth 1 -delete
        rmdir "$RELEASE_SRC"
        ;;
      *) fatal "拒绝清理非预期 RELEASE_SRC" ;;
    esac
  fi
}
trap cleanup_build_artifacts EXIT INT TERM

git archive --format=tar "$TARGET_COMMIT" > "$SOURCE_TAR"
test -s "$SOURCE_TAR"
test "$(git get-tar-commit-id < "$SOURCE_TAR")" = "$TARGET_COMMIT"
tar -xf "$SOURCE_TAR" -C "$RELEASE_SRC"
grep -q 'APP_VERSION = "1.19.0"' "$RELEASE_SRC/frontend/src/version.ts"

sudo docker compose \
  --project-name "$BUILD_PROJECT" \
  --env-file "$PROD_ENV" \
  --project-directory "$RELEASE_SRC" \
  -f "$RELEASE_SRC/docker-compose.yml" \
  build --pull app frontend

# 仅 build 后，`compose images -q` 依赖已创建容器，可能为空或指向旧容器。为两个新制品
# 创建“从不启动”的隔离探针容器，并从容器 `.Image` 取唯一内容 ID。
sudo docker compose \
  --project-name "$BUILD_PROJECT" \
  --env-file "$PROD_ENV" \
  --project-directory "$RELEASE_SRC" \
  -f "$RELEASE_SRC/docker-compose.yml" \
  create --no-build --pull never app frontend

BUILD_APP_CID=$(sudo docker compose \
  --project-name "$BUILD_PROJECT" \
  --env-file "$PROD_ENV" \
  --project-directory "$RELEASE_SRC" \
  -f "$RELEASE_SRC/docker-compose.yml" ps --all --quiet app)
BUILD_FRONTEND_CID=$(sudo docker compose \
  --project-name "$BUILD_PROJECT" \
  --env-file "$PROD_ENV" \
  --project-directory "$RELEASE_SRC" \
  -f "$RELEASE_SRC/docker-compose.yml" ps --all --quiet frontend)
test -n "$BUILD_APP_CID"
test -n "$BUILD_FRONTEND_CID"
test "$(sudo docker inspect -f '{{.State.Status}}' "$BUILD_APP_CID")" = "created"
test "$(sudo docker inspect -f '{{.State.Status}}' "$BUILD_FRONTEND_CID")" = "created"
test "$(sudo docker inspect -f '{{index .Config.Labels "com.docker.compose.project"}}' \
  "$BUILD_APP_CID")" = "$BUILD_PROJECT"
test "$(sudo docker inspect -f '{{index .Config.Labels "com.docker.compose.service"}}' \
  "$BUILD_APP_CID")" = "app"
test "$(sudo docker inspect -f '{{index .Config.Labels "com.docker.compose.service"}}' \
  "$BUILD_FRONTEND_CID")" = "frontend"

NEW_APP_IMAGE_ID=$(sudo docker inspect -f '{{.Image}}' "$BUILD_APP_CID")
NEW_FRONTEND_IMAGE_ID=$(sudo docker inspect -f '{{.Image}}' "$BUILD_FRONTEND_CID")
[[ "$NEW_APP_IMAGE_ID" == sha256:* ]]
[[ "$NEW_FRONTEND_IMAGE_ID" == sha256:* ]]
test "$NEW_APP_IMAGE_ID" != "$OLD_APP_IMAGE_ID"
test "$NEW_FRONTEND_IMAGE_ID" != "$OLD_FRONTEND_IMAGE_ID"

save_state SOURCE_TAR "$SOURCE_TAR"
save_state NEW_APP_IMAGE_ID "$NEW_APP_IMAGE_ID"
save_state NEW_FRONTEND_IMAGE_ID "$NEW_FRONTEND_IMAGE_ID"
save_state RELEASE_PHASE built
sha256sum "$SOURCE_TAR" > "${SOURCE_TAR}.sha256"
sha256sum -c "${SOURCE_TAR}.sha256"

cleanup_build_artifacts
test ! -e "$RELEASE_SRC"
trap - EXIT INT TERM
```

## 3. 最终备份、checksum、TOC、临时库恢复演练与旧状态指纹

### 3.1 备份必须真的能恢复

只看到 `pg_dump` 成功不算备份可用。必须校验 checksum、读取 TOC，并恢复进一次性临时库；
临时库的 Alembic head 和三张关键事实表行数必须与源库一致。

```bash
BACKUP="$PWD/backups/v1.19-pre-${TARGET_COMMIT}-$(date +%Y%m%d-%H%M%S).dump"
BACKUP_TOC="${BACKUP}.toc"
BACKUP_SUM="${BACKUP}.sha256"

sudo docker compose exec -T db pg_dump \
  -U spareparts -Fc --no-owner spareparts > "$BACKUP"
test -s "$BACKUP"
sudo docker compose exec -T db pg_restore --list < "$BACKUP" > "$BACKUP_TOC"
test -s "$BACKUP_TOC"
sha256sum "$BACKUP" "$BACKUP_TOC" > "$BACKUP_SUM"
sha256sum -c "$BACKUP_SUM"

SOURCE_COUNTS=$(psql_db -At -F '|' -c "
  SELECT
    (SELECT count(*) FROM f_maintenance_order),
    (SELECT count(*) FROM f_maintenance_line),
    (SELECT count(*) FROM f_sales_order);")

# 临时恢复会与生产库共用 PostgreSQL 卷。可用空间至少为源库大小 1.5 倍再加 2 GiB；
# 不足时立即停止，不得在生产卷演练。应把备份带到独立 PostgreSQL 15 实例恢复，
# 完成同样的 head/行数断言并归档实例与存储证据后，才能重新进入发布流程。
DB_SIZE_BYTES=$(psql_db -At -c "SELECT pg_database_size('spareparts');")
PGDATA_FREE_BYTES=$(sudo docker compose exec -T db sh -c \
  "df -PB1 /var/lib/postgresql/data | awk 'NR==2 {print \$4}'")
[[ "$DB_SIZE_BYTES" =~ ^[0-9]+$ ]]
[[ "$PGDATA_FREE_BYTES" =~ ^[0-9]+$ ]]
RESTORE_REQUIRED_BYTES=$((DB_SIZE_BYTES + DB_SIZE_BYTES / 2 + 2147483648))
test "$PGDATA_FREE_BYTES" -ge "$RESTORE_REQUIRED_BYTES" \
  || fatal "生产 PostgreSQL 卷空间不足，禁止在该卷创建恢复演练库；改用隔离 DB 实例"

RESTORE_DB="v119_restore_${$}_$(date +%s)"
[[ "$RESTORE_DB" =~ ^[a-z0-9_]+$ ]]
cleanup_restore_db() {
  psql_postgres -c "
    SELECT pg_terminate_backend(pid) FROM pg_stat_activity
    WHERE datname='$RESTORE_DB' AND pid <> pg_backend_pid();" >/dev/null
  psql_postgres -c "DROP DATABASE IF EXISTS \"$RESTORE_DB\";" >/dev/null
  test "$(psql_postgres -At -c \
    "SELECT count(*) FROM pg_database WHERE datname='$RESTORE_DB';")" = "0"
}
cleanup_restore_db_on_exit() {
  restore_exit_status=$?
  if ! cleanup_restore_db; then
    printf 'FATAL: 临时恢复库清理失败，保持应用停止并立即升级处理：%s\n' \
      "$RESTORE_DB" >&2
    restore_exit_status=97
  fi
  trap - EXIT INT TERM
  exit "$restore_exit_status"
}
trap cleanup_restore_db_on_exit EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
psql_postgres -c "CREATE DATABASE \"$RESTORE_DB\" OWNER spareparts;"
sudo docker compose exec -T db pg_restore \
  -U spareparts -d "$RESTORE_DB" --exit-on-error --no-owner < "$BACKUP"
RESTORED_HEAD=$(sudo docker compose exec -T db \
  psql -U spareparts -d "$RESTORE_DB" -v ON_ERROR_STOP=1 -At \
  -c "SELECT version_num FROM alembic_version;")
RESTORED_COUNTS=$(sudo docker compose exec -T db \
  psql -U spareparts -d "$RESTORE_DB" -v ON_ERROR_STOP=1 -At -F '|' -c "
    SELECT
      (SELECT count(*) FROM f_maintenance_order),
      (SELECT count(*) FROM f_maintenance_line),
      (SELECT count(*) FROM f_sales_order);")
test "$RESTORED_HEAD" = "$OLD_DB_HEAD"
test "$RESTORED_COUNTS" = "$SOURCE_COUNTS"
cleanup_restore_db
trap - EXIT INT TERM

save_state BACKUP "$BACKUP"
save_state BACKUP_TOC "$BACKUP_TOC"
save_state BACKUP_SUM "$BACKUP_SUM"
```

### 3.2 保存可跨迁移比较的成本基线和历史报销迁移证据

DEV-15 会重排成本瀑布，发布前已有来源也可能合法变化，因此不再保存“旧五层不可变化”
白名单。改为保存旧 schema 已存在的行级字段，发布后证明行集合不增不减、第一次变化可审计、第二次重算
完全幂等。同时保存历史报销的聚合证据；甲方已确认缺少明确税务字段的历史报销按未税解释。

```bash
PRE_COST="$PWD/backups/v1.19-cost-before-${TARGET_COMMIT}.csv"
PRE_SOURCE_COUNTS="$PWD/backups/v1.19-source-counts-before-${TARGET_COMMIT}.csv"
PRE_EXPENSE_AUDIT="$PWD/backups/v1.19-expense-before-${TARGET_COMMIT}.csv"
psql_db -At -F '|' -c "
  SELECT id, unit_cost, cost_amount, cost_source, cost_tax_basis
  FROM f_maintenance_line ORDER BY id;" > "$PRE_COST"
psql_db -At -F '|' -c "
  SELECT coalesce(cost_source,'<NULL>'), count(*)
  FROM f_maintenance_line
  GROUP BY coalesce(cost_source,'<NULL>')
  ORDER BY 1;" > "$PRE_SOURCE_COUNTS"
psql_db -At -F '|' -c "
  SELECT
    count(*) AS expense_rows,
    count(DISTINCT import_batch_id) AS expense_batches,
    count(*) FILTER (WHERE amount IS NULL) AS null_amount_rows,
    count(*) FILTER (WHERE amount < 0) AS negative_amount_rows
  FROM f_project_expense;" > "$PRE_EXPENSE_AUDIT"

test -f "$PRE_COST"
test -s "$PRE_SOURCE_COUNTS"
test -s "$PRE_EXPENSE_AUDIT"
sha256sum "$PRE_COST" "$PRE_SOURCE_COUNTS" "$PRE_EXPENSE_AUDIT" >> "$BACKUP_SUM"
sha256sum -c "$BACKUP_SUM"
save_state PRE_COST "$PRE_COST"
save_state PRE_SOURCE_COUNTS "$PRE_SOURCE_COUNTS"
save_state PRE_EXPENSE_AUDIT "$PRE_EXPENSE_AUDIT"
save_state RELEASE_PHASE backup_verified
```

备份、TOC、checksum、临时库恢复或旧状态指纹任一失败：不得继续。

### 3.3 迁移开始前失败：只恢复旧制品，不碰数据库

第 1 节失败时生产服务尚未停止，清理构建临时物后结束即可。第 2、3 节失败，或第 4 节在
`alembic upgrade head` **开始执行前**失败时，数据库仍必须精确处于 `OLD_DB_HEAD`；此时不得
进入第 11 节，也不得恢复任何备份。另开 shell 执行下列原地恢复，把可能只切了一半的镜像
引用和 checkout 一并恢复到旧基线：

```bash

set -Eeuo pipefail
umask 077
cd ~/apps/it-spareparts
source backups/v1.19-release-state.env
fatal() { printf 'FATAL: %s\n' "$*" >&2; exit 1; }
psql_db() {
  sudo docker compose exec -T db \
    psql -U spareparts -d spareparts -v ON_ERROR_STOP=1 "$@"
}

case "${RELEASE_PHASE:-}" in
  built|writes_closed|backup_verified) ;;
  *) fatal "状态已进入 ${RELEASE_PHASE:-unknown}，禁止使用迁移前恢复分支，改走第 11 节" ;;
esac
test "$(psql_db -At -c 'SELECT version_num FROM alembic_version;')" = "$OLD_DB_HEAD" \
  || fatal "数据库已不在 OLD_DB_HEAD，必须改走第 11 节"
git cat-file -e "${OLD_COMMIT}^{commit}"
test "$(sudo docker image inspect -f '{{.Id}}' "$OLD_APP_IMAGE_ID")" \
  = "$OLD_APP_IMAGE_ID"
test "$(sudo docker image inspect -f '{{.Id}}' "$OLD_FRONTEND_IMAGE_ID")" \
  = "$OLD_FRONTEND_IMAGE_ID"

PRE_MIGRATION_ABORT_COMPLETE=0
pre_migration_abort_fail_closed() {
  pre_abort_exit_status=$?
  if [[ "${PRE_MIGRATION_ABORT_COMPLETE:-0}" != "1" ]]; then
    sudo docker compose stop frontend app >/dev/null 2>&1 || true
    if pre_abort_running=$(
      sudo docker compose ps --status running --services app frontend 2>/dev/null
    ); then
      if [[ -n "$pre_abort_running" ]]; then
        printf 'FATAL: 迁移前恢复失败且服务仍在运行：%s\n' "$pre_abort_running" >&2
        pre_abort_exit_status=99
      else
        printf '迁移前恢复尚未验收，app/frontend 已保持停止。\n' >&2
      fi
    else
      printf 'FATAL: 迁移前恢复失败，且无法验证 app/frontend 是否停止。\n' >&2
      pre_abort_exit_status=99
    fi
    if [[ "$pre_abort_exit_status" = "0" ]]; then
      pre_abort_exit_status=98
    fi
  fi
  trap - EXIT INT TERM
  exit "$pre_abort_exit_status"
}
trap pre_migration_abort_fail_closed EXIT
trap 'exit 130' INT
trap 'exit 143' TERM

sudo docker compose stop frontend app
git checkout --detach "$OLD_COMMIT"
test "$(git rev-parse HEAD)" = "$OLD_COMMIT"
sudo docker tag "$OLD_APP_IMAGE_ID" "$APP_IMAGE_REF"
sudo docker tag "$OLD_FRONTEND_IMAGE_ID" "$FRONTEND_IMAGE_REF"
test "$(sudo docker image inspect -f '{{.Id}}' "$APP_IMAGE_REF")" \
  = "$OLD_APP_IMAGE_ID"
test "$(sudo docker image inspect -f '{{.Id}}' "$FRONTEND_IMAGE_REF")" \
  = "$OLD_FRONTEND_IMAGE_ID"

sudo docker compose up -d --no-deps --no-build --force-recreate app frontend
test "$(sudo docker inspect -f '{{.Image}}' "$(sudo docker compose ps -q app)")" \
  = "$OLD_APP_IMAGE_ID"
test "$(sudo docker inspect -f '{{.Image}}' "$(sudo docker compose ps -q frontend)")" \
  = "$OLD_FRONTEND_IMAGE_ID"
sudo docker compose exec -T app python - <<'PY'
import json
from urllib.request import urlopen
for path in ("/health", "/health/db"):
```

失败恢复矩阵：

| 失败位置 | 可证明的现场状态 | 唯一恢复路径 |
| --- | --- | --- |
| 第 1 节 | 旧服务仍运行、DB 未变 | 清理隔离构建资源，保留旧服务 |
| 第 2–3 节 | 服务已停、DB 仍为 `OLD_DB_HEAD` | 本节原地恢复旧制品 |
| 第 4 节迁移命令开始前 | DB 仍旧，镜像引用/checkout 可能半切换 | 本节重打旧 tag、恢复旧 SHA |
| 第 4 节迁移开始后 | DB 可能为 OLD、d4、e5 或 NEW | 第 11 节先事故备份，再精确 downgrade |
| 第 5–10 节 | NEW head，重算或业务写入可能已提交 | 第 11 节保写回滚，禁止恢复发布前备份 |

### 6.3 用独立 SQL 计算期望毛利，再与应用 board 对账

下面的 SQL 不调用应用的毛利函数。它按“成功批次中的最新已生效销售事实”选择收入，固定以
13% 生成含税值，并独立计算备件毛利及费用水位完整时的贡献毛利。至少必须有一个可对账合同；
否则生产快照不足以验收本功能。

```bash
SQL_MARGIN="$PWD/backups/v1.19-independent-margin-${TARGET_COMMIT}.csv"
psql_db -At -F '|' -c "
  WITH scoped_cost AS (
    SELECT coalesce(o.linked_sales_order_no,'') AS contract,
           count(*) AS lines,
           count(*) FILTER (
             WHERE l.cost_source IS NULL OR l.cost_source='none'
                OR l.cost_amount_inc_tax IS NULL OR l.cost_amount_ex_tax IS NULL
           ) AS missing_lines,
           round(sum(l.cost_amount_inc_tax),2) AS cost_inc,
           round(sum(l.cost_amount_ex_tax),2) AS cost_ex
    FROM f_maintenance_line l
    JOIN f_maintenance_order o ON o.id=l.order_id
    WHERE o.data_status='已生效' AND o.order_date >= DATE '2024-01-01'
    GROUP BY coalesce(o.linked_sales_order_no,'')
  ),
  ranked_revenue AS (
    SELECT s.order_no,
           s.amount_ex_tax,
           row_number() OVER (
             PARTITION BY s.order_no
             ORDER BY b.uploaded_at DESC NULLS LAST,
                      s.created_at DESC NULLS LAST,
                      s.import_batch_id DESC,
                      s.id DESC
           ) AS version_rank
    FROM f_sales_order s
    JOIN sys_import_batch b ON b.id=s.import_batch_id
    WHERE s.data_status='已生效'
      AND b.status='success'
  ),
  revenue AS (
    SELECT order_no, amount_ex_tax AS revenue_ex
    FROM ranked_revenue
    WHERE version_rank=1
  ),
  expense AS (
    SELECT linked_sales_order_no AS contract,
           round(sum(amount_ex_tax),2) AS expense_ex,
           round(sum(amount_inc_tax),2) AS expense_inc,
           count(*) FILTER (
             WHERE amount_ex_tax IS NULL OR amount_inc_tax IS NULL
           ) AS missing_expense
    FROM f_project_expense
    WHERE data_status='已结束'
      AND linked_sales_order_no IS NOT NULL
      AND linked_sales_order_no <> ''
    GROUP BY linked_sales_order_no
  ),
  normalized AS (
    SELECT c.contract,
           round(r.revenue_ex,2) AS revenue_ex,
           round(r.revenue_ex*1.13,2) AS revenue_inc,
           c.cost_ex,
           c.cost_inc,
           (
             coalesce(ws.expense_snapshot_complete,false)
             AND coalesce(e.missing_expense,0)=0
           ) AS expense_ready,
           CASE
             WHEN coalesce(ws.expense_snapshot_complete,false)
               AND coalesce(e.missing_expense,0)=0
             THEN coalesce(e.expense_ex,0)
           END AS expense_ex,
           CASE
             WHEN coalesce(ws.expense_snapshot_complete,false)
               AND coalesce(e.missing_expense,0)=0
             THEN coalesce(e.expense_inc,0)
           END AS expense_inc
    FROM scoped_cost c
    JOIN revenue r ON r.order_no=c.contract
    LEFT JOIN maintenance_contract_workbook_state ws
      ON ws.contract_no=c.contract
    LEFT JOIN expense e ON e.contract=c.contract
    WHERE c.contract <> ''
      AND c.lines > 0 AND c.missing_lines=0
      AND r.revenue_ex IS NOT NULL AND r.revenue_ex > 0
  ),
  expected AS (
    SELECT contract, revenue_ex, revenue_inc, cost_ex, cost_inc, expense_ready,
           expense_ex, expense_inc,
           round(revenue_ex-cost_ex,2) AS profit_ex,
           round(revenue_inc-cost_inc,2) AS profit_inc,
           round((revenue_ex-cost_ex)/revenue_ex,4) AS margin_ex,
           round((revenue_inc-cost_inc)/revenue_inc,4) AS margin_inc,
           CASE WHEN expense_ready
             THEN round(revenue_ex-cost_ex-expense_ex,2) END AS contribution_ex,
           CASE WHEN expense_ready
             THEN round(revenue_inc-cost_inc-expense_inc,2) END AS contribution_inc,
           CASE WHEN expense_ready
             THEN round((revenue_ex-cost_ex-expense_ex)/revenue_ex,4)
           END AS contribution_margin_ex,
           CASE WHEN expense_ready
             THEN round((revenue_inc-cost_inc-expense_inc)/revenue_inc,4)
           END AS contribution_margin_inc
    FROM normalized
    WHERE revenue_inc > 0
  )
  SELECT contract, revenue_ex, revenue_inc, cost_ex, cost_inc,
         profit_ex, profit_inc, margin_ex, margin_inc,
         expense_ready, expense_ex, expense_inc,
         contribution_ex, contribution_inc,
         contribution_margin_ex, contribution_margin_inc
  FROM expected ORDER BY contract;" > "$SQL_MARGIN"
test -s "$SQL_MARGIN" || fatal "没有可用于独立毛利对账的合同"
chmod 600 "$SQL_MARGIN"
```

```bash
    fi
    if [[ "$rollback_exit_status" = "0" ]]; then

    fi
  fi
  trap - EXIT INT TERM
  exit "$rollback_exit_status"
```

## 12. 灾难恢复边界

发布前 `BACKUP` 的用途是：

- 开放生产入口之前，数据库发生无法通过迁移修复的物理损坏；
- 恢复演练和最终灾难取证。

一旦在启动入口前记录了 `OPENED_AT`/`RELEASE_PHASE=opening`，就按“可能已有真实业务写入”
处理，发布前备份不能再作为正常回滚源。若数据库已
物理损坏，优先恢复第 11 节的 `INCIDENT_BACKUP`；只有事故后备份也不可用时，才可由业务负责人
和数据库负责人批准“恢复发布前备份 + 逐笔补录/对账”的数据恢复方案。该方案属于数据事故，
不属于本 Runbook 的自动回滚。
